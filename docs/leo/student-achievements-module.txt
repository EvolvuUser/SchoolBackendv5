What column is for what: 

title → Short headline of achievement
👉 Example: "Won 1st Prize in Hackathon"
description → Detailed explanation
👉 Example: "Built an AI chatbot in 24 hours and won first place among 30 teams"
event_name → Name of event/competition
👉 Example: "TechFest 2026"
organization_name → Who conducted it
👉 Example: "IIT Bombay"

type_id → Type/category of achievement
👉 Example:
Academic
Sports
Cultural
Technical
NCC/NSS etc.
level_id → Level of achievement
👉 Example:
School
District
State
National
International

score → Any score/value
👉 Example:
"95%"
"8.5 CGPA"
"Gold Medal"
position → Rank achieved
👉 Example:
1, 2, 3
or NULL if not applicable

Tables:

-------------------------------- QUERIES TO RUN 

CREATE TABLE student_achievements (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    reg_no VARCHAR(10) NULL,
    student_id BIGINT UNSIGNED NOT NULL,
    academic_year VARCHAR(11) NULL,
    title VARCHAR(255) NOT NULL,
    description TEXT NULL,
    achievement_date DATE NOT NULL,
    type VARCHAR(20),
    level VARCHAR(20),
    organization_name VARCHAR(255) NULL,
    event_name VARCHAR(255) NULL,
    score VARCHAR(50) NULL,
    position INT NULL,
    is_verified TINYINT(1) DEFAULT 0,
    is_external TINYINT(1) DEFAULT 1,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE achievement_files (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    achievement_id BIGINT UNSIGNED NOT NULL,
    file_url TEXT NOT NULL,
    file_type VARCHAR(20) NULL,
    uploaded_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
